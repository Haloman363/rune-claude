# OSRS TUI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a standalone Python Textual TUI that replicates the OSRS Fixed Mode game interface with placeholders throughout.

**Architecture:** A Textual `App` composes four custom widgets — `GameViewport`, `Minimap`, `ControlPanel`, `Chatbox` — laid out using Textual's CSS grid to match OSRS Fixed Mode proportions. Each widget is self-contained in its own file. Styling lives in a single `osrs.tcss` file.

**Tech Stack:** Python 3.8+, `textual` (pip install textual), stdlib only otherwise.

```python
    "..T..........T.............~~~~~.............................",
    "...........T.....T.........~~~~~....T.....T.................",
]

_PLAYER_ROW = len(_WORLD) // 2
_PLAYER_COL = len(_WORLD[0]) // 2


class GameViewport(Widget):
    """Static OSRS-styled ASCII tile world (Phase 1 placeholder)."""

    DEFAULT_CSS = """
    GameViewport {
        border: solid #605443;
        background: #18140c;
        padding: 0 1;
    }
    """

    def render(self) -> RenderResult:
        text = Text()
        for row_idx, row in enumerate(_WORLD):
            for col_idx, ch in enumerate(row):
                if row_idx == _PLAYER_ROW and col_idx == _PLAYER_COL:
                    text.append("@", style="bold bright_white")
                elif ch == "T":
                    text.append("T", style="#228B22")  # tree green
                elif ch == "^":
                    text.append("^", style="#888888")  # rock grey
                elif ch == "~":
                    text.append("~", style="#1e90ff")  # water blue
                else:
                    text.append(".", style="#3a5c1a")  # grass dark green
            text.append("\n")
        return text
```

- [ ] **Step 3: Write `tui/widgets/panel.py`**

```python
from textual.app import ComposeResult
from textual.widget import Widget
from textual.widgets import Button, Static
from textual.reactive import reactive

_TABS_ROW1 = ["Combat", "Skills", "Quest", "Inventory", "Equipment", "Prayer", "Magic"]
_TABS_ROW2 = ["Clan", "Friends", "Account", "Logout", "Settings", "Emotes", "Music"]
_ALL_TABS = _TABS_ROW1 + _TABS_ROW2


class ControlPanel(Widget):
    """Right control panel — 14 tabs + body (Phase 1: inventory grid or placeholder)."""

    active_tab: reactive[str] = reactive("Inventory")

    DEFAULT_CSS = """
    ControlPanel {
        layout: vertical;
        background: #2a2316;
    }
    .tab-row {
        layout: horizontal;
        height: 1;
        background: #18140c;
    }
    .tab-btn {
        background: #2a2316;
        color: #c0a886;
        border: none;
        padding: 0 1;
        height: 1;
        min-width: 0;
        width: 1fr;
    }
    .tab-btn.-active {
        color: #ffcc00;
        text-style: underline;
    }
    #panel-body {
        height: 1fr;
    }
    .inventory-grid {
        layout: grid;
        grid-size: 4 7;
        padding: 1;
        height: 1fr;
    }
    .item-slot {
        border: solid #605443;
        background: #18140c;
        height: 3;
        content-align: center middle;
        color: #605443;
    }
    .panel-placeholder {
        color: #c0a886;
        content-align: center middle;
        height: 1fr;
    }
    """

    def set_active_tab(self, tab: str) -> None:
        if tab not in _ALL_TABS:
            raise ValueError(f"Unknown tab: {tab!r}. Valid: {_ALL_TABS}")
        self.active_tab = tab

    def compose(self) -> ComposeResult:
        with Static(classes="tab-row"):
            for tab in _TABS_ROW1:
                btn = Button(tab, classes="tab-btn", id=f"tab-{tab.lower()}")
                if tab == self.active_tab:
                    btn.add_class("-active")
                yield btn
        with Static(classes="tab-row"):
            for tab in _TABS_ROW2:
                btn = Button(tab, classes="tab-btn", id=f"tab-{tab.lower()}")
                if tab == self.active_tab:
                    btn.add_class("-active")
                yield btn
        yield self._make_body()

    def _make_body(self) -> Widget:
        if self.active_tab == "Inventory":
            grid = Static(id="panel-body", classes="inventory-grid")
            return grid
        return Static(
            f"[ {self.active_tab} ]",
            id="panel-body",
            classes="panel-placeholder",
        )

    def on_button_pressed(self, event: Button.Pressed) -> None:
        tab_name = event.button.label.plain
        if tab_name in _ALL_TABS:
            self.set_active_tab(tab_name)
            # Refresh tab highlights
            for btn in self.query(".tab-btn"):
                if btn.label.plain == tab_name:
                    btn.add_class("-active")
                else:
                    btn.remove_class("-active")
            # Replace panel body
            old_body = self.query_one("#panel-body")
            old_body.remove()
            self.mount(self._make_body())

    def on_mount(self) -> None:
        """Populate inventory grid with 28 item slots after mount."""
        if self.active_tab == "Inventory":
            try:
                grid = self.query_one("#panel-body")
                for _ in range(28):
                    grid.mount(Static("[ ]", classes="item-slot"))
            except Exception:
                pass
```

```python
from dataclasses import dataclass
from textual.app import ComposeResult
from textual.widget import Widget
from textual.widgets import Button, Input, Static
from textual.reactive import reactive
from rich.text import Text

# Message style constants (CSS color strings)
MSG_GAME    = "white"
MSG_PUBLIC  = "#0000ff"
MSG_PRIVATE = "#7f0000"
MSG_XP      = "#ffcc00"
MSG_QUEST   = "#9900cc"

_CHAT_TABS = ["All", "Game", "Public", "Private", "Channel", "Clan", "Trade"]


@dataclass
class ChatMessage:
    text: str
    style: str = MSG_GAME


_INITIAL_MESSAGES = [
    ChatMessage("Welcome to Old School RuneScape.", MSG_GAME),
    ChatMessage("[Game] rune-claude TUI v0.1 loaded.", MSG_GAME),
    ChatMessage("[Game] Type a message and press Enter.", MSG_GAME),
]


class Chatbox(Widget):
    """OSRS chatbox — 7 tabs, 8 message lines, text input (Phase 1)."""

    active_tab: reactive[str] = reactive("All")

    DEFAULT_CSS = """
    Chatbox {
        layout: vertical;
        background: #2a2316;
        border: solid #605443;
    }
    .chat-tab-row {
        layout: horizontal;
        height: 1;
        background: #18140c;
    }
    .chat-tab-btn {
        background: #2a2316;
        color: #c0a886;
        border: none;
        height: 1;
        min-width: 0;
        width: auto;
        padding: 0 1;
    }
    .chat-tab-btn.-active {
        color: #ffcc00;
        text-style: underline;
    }
    #chat-messages {
        height: 8;
        overflow-y: auto;
        padding: 0 1;
        background: #18140c;
    }
    .chat-input-bar {
        height: 1;
        layout: horizontal;
        background: #18140c;
        border-top: solid #605443;
        padding: 0 1;
    }
    #chat-input-label {
        color: #ffcc00;
        width: auto;
    }
    #chat-input {
        background: #18140c;
        color: white;
        border: none;
        width: 1fr;
    }
    """

    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        self.messages: list[ChatMessage] = list(_INITIAL_MESSAGES)

    def set_active_tab(self, tab: str) -> None:
        if tab not in _CHAT_TABS:
            raise ValueError(f"Unknown tab: {tab!r}")
        self.active_tab = tab

    def add_message(self, text: str, style: str = MSG_PUBLIC) -> None:
        self.messages.append(ChatMessage(text, style))
        try:
            self._refresh_messages()
        except Exception:
            pass  # Widget may not be mounted yet

    def _refresh_messages(self) -> None:
        log = self.query_one("#chat-messages", Static)
        rendered = Text()
        for msg in self.messages[-8:]:
            rendered.append(msg.text + "\n", style=msg.style)
        log.update(rendered)

    def compose(self) -> ComposeResult:
        with Static(classes="chat-tab-row"):
            for tab in _CHAT_TABS:
                btn = Button(tab, classes="chat-tab-btn", id=f"ctab-{tab.lower()}")
                if tab == self.active_tab:
                    btn.add_class("-active")
                yield btn
        # Build initial message text
        initial = Text()
        for msg in self.messages[-8:]:
            initial.append(msg.text + "\n", style=msg.style)
        yield Static(initial, id="chat-messages")
        with Static(classes="chat-input-bar"):
            yield Static("Adventurer: ", id="chat-input-label")
            yield Input(placeholder="", id="chat-input")

    def on_button_pressed(self, event: Button.Pressed) -> None:
        label = event.button.label.plain
        if label in _CHAT_TABS:
            self.set_active_tab(label)
            for btn in self.query(".chat-tab-btn"):
                if btn.label.plain == label:
                    btn.add_class("-active")
                else:
                    btn.remove_class("-active")

    def on_input_submitted(self, event: Input.Submitted) -> None:
        text = event.value.strip()
        if text:
            self.add_message(text, MSG_PUBLIC)
            event.input.value = ""
```

```python
#!/usr/bin/env python3
"""
rune-claude OSRS TUI — Phase 1 placeholder layout.
Run with: python3 tui/main.py
Requires: pip install textual
"""
import sys
from pathlib import Path

# Allow running from repo root without installing the package
sys.path.insert(0, str(Path(__file__).parent.parent))

from textual.app import App, ComposeResult
from textual.widgets import Static
from textual import on
from textual.events import Resize

from tui.widgets.viewport import GameViewport
from tui.widgets.minimap import Minimap
from tui.widgets.panel import ControlPanel
from tui.widgets.chatbox import Chatbox

MIN_WIDTH  = 120
MIN_HEIGHT = 40

_SIZE_WARNING = (
    f"[yellow]Terminal too small.[/yellow]\n"
    f"Minimum size: {MIN_WIDTH}×{MIN_HEIGHT}\n"
    f"Please resize your terminal and relaunch."
)


class RuneClaudeTUI(App):
    """OSRS Fixed Mode TUI layout."""

    CSS_PATH = Path(__file__).parent / "styles" / "osrs.tcss"
    TITLE = "rune-claude"
    SUB_TITLE = "OSRS TUI v0.1"

    BINDINGS = [
        ("q", "quit", "Quit"),
        ("ctrl+c", "quit", "Quit"),
    ]

    def compose(self) -> ComposeResult:
        with Static(id="left-column"):
            yield GameViewport(id="game-viewport")
            yield Chatbox(id="chatbox")
        with Static(id="right-column"):
            yield Minimap(id="minimap")
            yield ControlPanel(id="control-panel")

    def on_mount(self) -> None:
        self._check_size()

    def on_resize(self, event: Resize) -> None:
        self._check_size()

    def _check_size(self) -> None:
        size = self.app.size
        if size.width < MIN_WIDTH or size.height < MIN_HEIGHT:
            self.notify(
                f"Terminal too small ({size.width}×{size.height}). "
                f"Minimum: {MIN_WIDTH}×{MIN_HEIGHT}",
                severity="warning",
                timeout=5,
            )


def main() -> None:
    app = RuneClaudeTUI()
    app.run()


if __name__ == "__main__":
    main()
```
